function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var raw = e.postData ? e.postData.contents : "{}";
    var data = JSON.parse(raw);

    if (data.action === "saveConfig" && data.config) {
      var cfgSheet = ss.getSheetByName("Portal_Config");
      if (!cfgSheet) {
        cfgSheet = ss.insertSheet("Portal_Config");
      }
      cfgSheet.clear();
      cfgSheet.getRange("A1:B1").setValues([["Updated_At", "Config_JSON"]]).setFontWeight("bold");
      cfgSheet.getRange("A2:B2").setValues([[new Date().toLocaleString("id-ID"), JSON.stringify(data.config)]]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "saveConfig" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "deleteEntry" && data.id) {
      var mainSheet = ss.getSheets()[0];
      var rows = mainSheet.getDataRange().getValues();
      for (var r = rows.length - 1; r >= 1; r--) {
        if (String(rows[r][0]).trim() === String(data.id).trim()) {
          mainSheet.deleteRow(r + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "deleteEntry", id: data.id }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var sheet = ss.getSheets()[0];
    var headers = [
      "ID", "Hari & Tanggal", "Periode", "Gelombang Audit",
      "Mode Identitas", "Nama Pengisi", "Inti / Biro / Divisi",
      "Jumlah Pertanyaan", "Rincian Pertanyaan & Jawaban",
      "Total Karakter", "Total Kata", "JSON_RAW"
    ];
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#D2E3D8");
    }
    var answers = data.answers || [];
    var formattedQA = answers.map(function(item, idx) {
      return "[" + (idx + 1) + "] " + item.question + "\nJawaban: " + item.answer;
    }).join("\n\n");

    sheet.appendRow([
      data.id || "",
      data.timestamp || new Date().toLocaleString("id-ID"),
      data.periodeName || "",
      data.waveTitle || "",
      data.isAnonymous ? "Anonim" : "Teridentifikasi",
      data.nama || "Anonim",
      data.divisi || "",
      answers.length,
      formattedQA,
      data.totalChars || 0,
      data.totalWords || 0,
      JSON.stringify(answers)
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", id: data.id }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  var values = sheet.getDataRange().getValues();
  var results = [];
  for (var i = 1; i < values.length; i++) {
    var r = values[i];
    var hasAnyValue = r.some(function(cell) {
      return String(cell || "").trim() !== "";
    });
    if (!hasAnyValue) continue;
    var idVal = String(r[0] || "").trim();
    var divVal = String(r[6] || "").trim();
    if (!idVal || !divVal) continue;

    var parsedAnswers = [];
    try {
      if (r[11]) parsedAnswers = JSON.parse(r[11]);
    } catch (err) {}
    if (!Array.isArray(parsedAnswers) || parsedAnswers.length === 0) {
      var fallbackAns = String(r[8] || "").trim();
      if (!fallbackAns) continue;
      parsedAnswers = [{ question: "Rincian Audit", answer: fallbackAns }];
    }
    results.push({
      id: idVal,
      timestamp: String(r[1] || ""),
      periodeName: String(r[2] || ""),
      waveTitle: String(r[3] || "Audit 1"),
      isAnonymous: String(r[4]) === "Anonim",
      nama: String(r[5] || "Anonim"),
      divisi: divVal,
      answers: parsedAnswers
    });
  }

  var savedConfig = null;
  var cfgSheet = ss.getSheetByName("Portal_Config");
  if (cfgSheet && cfgSheet.getLastRow() >= 2) {
    try {
      var rawCfg = cfgSheet.getRange("B2").getValue();
      if (rawCfg) savedConfig = JSON.parse(rawCfg);
    } catch (err) {}
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    data: results.reverse(),
    config: savedConfig
  })).setMimeType(ContentService.MimeType.JSON);
}
