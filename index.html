Langkah Singkat Pemasangan:

  1. Buka Google Sheets baru → klik menu Ekstensi → Apps Script.
  2. Hapus kode lama, tempel kode di bawah, lalu klik Terapkan → Deployment baru.
  3. Pilih jenis Aplikasi Web, atur Jalankan sebagai: Saya (Me), dan Yang memiliki akses: Siapa saja (Anyone).
  4. Salin URL Web App (berakhiran /exec) lalu tempel ke kolom di tab Pengaturan.

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var headers = [
      "ID", "Hari & Tanggal", "Periode", "Gelombang Audit",
      "Mode Identitas", "Nama Pengisi", "Inti / Biro / Divisi",
      "Jumlah Pertanyaan", "Rincian Pertanyaan & Jawaban",
      "Total Karakter", "Total Kata", "JSON_RAW"
    ];
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#F3EBF9");
    }
    var raw = e.postData ? e.postData.contents : "{}";
    var data = JSON.parse(raw);
    var answers = data.answers || [];
    var formattedQA = answers.map(function(item, idx) {
      return "[" + (idx + 1) + "] " + item.question + "\nJawaban: " + item.answer;
    }).join("\n\n");

    sheet.appendRow([
      data.id || "",
      data.timestamp || new Date().toLocaleString("id-ID"),
      data.periodeName || "",
      data.waveTitle || "",
      data.isAnonymous ? "Anonim" : "Teridentifikasi",
      data.nama || "Anonim",
      data.divisi || "",
      answers.length,
      formattedQA,
      data.totalChars || 0,
      data.totalWords || 0,
      JSON.stringify(answers)
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", id: data.id }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var values = sheet.getDataRange().getValues();
  var results = [];
  for (var i = 1; i < values.length; i++) {
    var r = values[i];
    if (!r[0] && !r[6]) continue;
    var parsedAnswers = [];
    try {
      if (r[11]) parsedAnswers = JSON.parse(r[11]);
    } catch (err) {}
    if (!Array.isArray(parsedAnswers) || parsedAnswers.length === 0) {
      parsedAnswers = [{ question: "Rincian Audit", answer: String(r[8] || "") }];
    }
    results.push({
      id: String(r[0] || "AUD-" + i),
      timestamp: String(r[1] || ""),
      periodeName: String(r[2] || ""),
      waveTitle: String(r[3] || "Audit 1"),
      isAnonymous: String(r[4]) === "Anonim",
      nama: String(r[5] || "Anonim"),
      divisi: String(r[6] || ""),
      answers: parsedAnswers
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: results.reverse() }))
    .setMimeType(ContentService.MimeType.JSON);
}
